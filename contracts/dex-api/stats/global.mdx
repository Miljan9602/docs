---
title: "Global Stats"
description: "Aggregate statistics for the DEX"
api: "GET https://sei-api.dragonswap.app/api/v1/stats"
---

Returns aggregate statistics for the DragonSwap DEX, including total volume, liquidity, swap counts, user counts, 24-hour metrics, and hourly and daily history.

Statistics are split into four buckets with the same structure: `v2` (DEX V1 constant-product pools), `v3` (DEX V2 concentrated-liquidity pools), `rfq` (RFQ trades), and `all`. Use `all` for DEX-wide totals; its volume, liquidity, swap, transaction, and fee figures are the sums of the other three buckets.

This endpoint requires no parameters.

<ResponseField name="status" type="string">
  Request status indicator.
</ResponseField>

<ResponseField name="stats" type="object">
  Container for DEX statistics, keyed by bucket.

  <Expandable title="stats">
    <ResponseField name="v2" type="object">
      Statistics for `v2` pools. `v2` = DEX V1 (constant-product) pool; `v3` = DEX V2 (concentrated-liquidity) pool.

      <Expandable title="Bucket object">
        <ResponseField name="total_volume_usd" type="number">
          Total cumulative trading volume in USD.
        </ResponseField>
        <ResponseField name="total_liquidity_usd" type="number">
          Current total liquidity in USD.
        </ResponseField>
        <ResponseField name="total_number_of_swaps" type="number">
          Total number of swaps executed.
        </ResponseField>
        <ResponseField name="total_users_count" type="number">
          Total number of unique users. The same value is reported in every bucket.
        </ResponseField>
        <ResponseField name="24_hour_tx_count" type="number">
          Number of transactions in the last 24 hours.
        </ResponseField>
        <ResponseField name="24_hour_fees_usd" type="number">
          Fees collected in the last 24 hours in USD.
        </ResponseField>
        <ResponseField name="24_hour_volume_usd" type="number">
          Trading volume in the last 24 hours in USD.
        </ResponseField>
        <ResponseField name="hourly_data" type="array">
          Hourly entries covering the last 6 months, oldest first.

          <Expandable title="hourly_data item">
            <ResponseField name="created_at" type="number">
              Start of the hour, as a Unix timestamp (seconds).
            </ResponseField>
            <ResponseField name="volume_usd" type="number">
              Trading volume in USD for the hour.
            </ResponseField>
            <ResponseField name="total_liquidity_usd" type="number">
              Total liquidity in USD at the snapshot time.
            </ResponseField>
            <ResponseField name="inflows_usd" type="number">
              Liquidity inflows in USD for the hour.
            </ResponseField>
            <ResponseField name="outflows_usd" type="number">
              Liquidity outflows in USD for the hour.
            </ResponseField>
            <ResponseField name="fees_usd" type="number">
              Fees collected in USD for the hour.
            </ResponseField>
          </Expandable>
        </ResponseField>
        <ResponseField name="daily_data" type="array">
          One entry per UTC day since the bucket's first recorded day, oldest first. The last entry is the current, still-incomplete day.

          <Expandable title="daily_data item">
            <ResponseField name="created_at" type="number">
              Start of the day (00:00 UTC), as a Unix timestamp (seconds).
            </ResponseField>
            <ResponseField name="volume_usd" type="number">
              Trading volume in USD for the day.
            </ResponseField>
            <ResponseField name="total_liquidity_usd" type="number">
              Total liquidity in USD at the snapshot time.
            </ResponseField>
            <ResponseField name="inflows_usd" type="number">
              Liquidity inflows in USD for the day.
            </ResponseField>
            <ResponseField name="outflows_usd" type="number">
              Liquidity outflows in USD for the day.
            </ResponseField>
            <ResponseField name="fees_usd" type="number">
              Fees collected in USD for the day.
            </ResponseField>
          </Expandable>
        </ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="v3" type="object">
      Statistics for `v3` pools. Same structure as `v2`.
    </ResponseField>

    <ResponseField name="rfq" type="object">
      Statistics for RFQ trades. Same structure as `v2`. `total_liquidity_usd`, `inflows_usd`, and `outflows_usd` are `0` for this bucket.
    </ResponseField>

    <ResponseField name="all" type="object">
      DEX-wide totals across `v2`, `v3`, and `rfq`. Same structure as `v2`.
    </ResponseField>
  </Expandable>
</ResponseField>
