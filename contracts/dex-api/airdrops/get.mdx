---
title: "Get Airdrop"
description: "Returns details for a specific airdrop campaign"
api: "GET https://sei-api.dragonswap.app/api/v1/airdrops/{airdrop}"
---

## Path Parameters

<ParamField path="airdrop" type="string" required default="0xe418e8f4245de5e03456b36cb8ff4158f1324f90">
  Airdrop contract address (case-insensitive). An unknown or malformed address returns `404` with `{"status":"fail"}`.
</ParamField>

## Response Fields

<ResponseField name="status" type="string">
  `"ok"` on success, `"fail"` when the airdrop is not found.
</ResponseField>

<ResponseField name="airdrop" type="object">
  Airdrop campaign details.

  <Expandable title="Airdrop object properties">
    <ResponseField name="name" type="string">
      Human-readable name of the airdrop campaign.
    </ResponseField>

    <ResponseField name="address" type="string">
      Contract address of the airdrop.
    </ResponseField>

    <ResponseField name="token_address" type="string">
      Contract address of the token being airdropped.
    </ResponseField>

    <ResponseField name="number_of_users" type="number">
      Total number of eligible users for the airdrop.
    </ResponseField>

    <ResponseField name="timestamps" type="array">
      Vesting schedule milestones, as Unix timestamps (seconds). One entry per vesting period; the same periods as `amounts` in [User Airdrop Allocation](/contracts/dex-api/airdrops/user).
    </ResponseField>

    <ResponseField name="vesting_text" type="string">
      Human-readable description of the vesting schedule.
    </ResponseField>

    <ResponseField name="total_claimed" type="number">
      Total amount of tokens claimed so far.
    </ResponseField>

    <ResponseField name="total_deposited" type="number">
      Total amount of tokens deposited into the airdrop contract.
    </ResponseField>

    <ResponseField name="starts_at" type="number">
      Time the airdrop becomes claimable, as a Unix timestamp (seconds).
    </ResponseField>

    <ResponseField name="ends_at" type="number">
      Time the airdrop expires, as a Unix timestamp (seconds).
    </ResponseField>
  </Expandable>
</ResponseField>
