---
title: "Staking Stats"
description: "Historical staking statistics across multiple time periods"
api: "GET https://sei-api.dragonswap.app/api/v1/staking/{address}/stats"
---

## Path Parameters

<ParamField path="address" type="string" required default="0xb71886c52d754cf2b1d3d866c7cfe9eec01418a7">
  Staking contract address. Any other address returns `"total_staked": 0` and empty arrays.
</ParamField>

## Response Fields

<ResponseField name="status" type="string">
  `"ok"` on success
</ResponseField>

<ResponseField name="total_staked" type="number">
  Total DRG currently staked (human-readable). Equals `total_deposits` of the latest snapshot.
</ResponseField>

<ResponseField name="stats" type="object">
  Staking snapshots grouped by time period. Every entry has the same shape.

  <Expandable title="Stats object properties">
    <ResponseField name="daily" type="object[]">
      Snapshots from the last 24 hours, taken about every 10 minutes (about 144 entries), oldest first

      <Expandable title="Snapshot properties">
        <ResponseField name="total_deposits" type="number">
          Total DRG staked at the snapshot (human-readable)
        </ResponseField>

        <ResponseField name="total_deposits_usd" type="number">
          USD value of `total_deposits`
        </ResponseField>

        <ResponseField name="total_fees" type="number">
          Total fees collected
        </ResponseField>

        <ResponseField name="apr" type="number">
          Staking APR, in percent
        </ResponseField>

        <ResponseField name="created_at" type="number">
          Snapshot time, Unix timestamp (seconds)
        </ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="weekly" type="object[]">
      Weekly snapshots (same shape as `daily`). Currently contains a single entry: the latest snapshot.
    </ResponseField>

    <ResponseField name="monthly" type="object[]">
      Monthly snapshots (same shape as `daily`). Currently contains a single entry: the latest snapshot.
    </ResponseField>

    <ResponseField name="all_time" type="object[]">
      Long-range snapshots at irregular intervals, oldest first, ending with the latest snapshot (same shape as `daily`). The earliest entry is currently from 2025-11-20.
    </ResponseField>
  </Expandable>
</ResponseField>
