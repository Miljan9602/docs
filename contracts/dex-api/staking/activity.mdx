---
title: "Staking Activity"
description: "A user's staking activity including rewards, events, marketplace actions, and bonuses"
api: "GET https://sei-api.dragonswap.app/api/v1/staking/{address}/activity/{user}"
---

All arrays are sorted newest first. All amounts are strings in the token's smallest unit (DRG: 18 decimals, USDC: 6 decimals) unless noted, and all timestamps are strings.

## Path Parameters

<ParamField path="address" type="string" required default="0xb71886c52d754cf2b1d3d866c7cfe9eec01418a7">
  Staking contract address. Any other address returns `404` (`staker_address_not_found`).
</ParamField>

<ParamField path="user" type="string" required>
  User wallet address (case-insensitive). A user with no staking history returns `200` with four empty arrays.
</ParamField>

## Response Fields

<ResponseField name="status" type="string">
  `"ok"` on success, `"fail"` on error
</ResponseField>

<ResponseField name="data" type="object">
  User activity data

  <Expandable title="Data object properties">
    <ResponseField name="reward_activity" type="object[]">
      Reward claims and compounding

      <Expandable title="Reward activity properties">
        <ResponseField name="type" type="string">
          `"Claim"`, `"Compound"`, `"Autocompound"`, or `"Autocompound Update"`
        </ResponseField>

        <ResponseField name="user" type="string">
          User wallet address
        </ResponseField>

        <ResponseField name="token" type="string | null">
          Reward token: `"1"` = DRG, `"2"` = USDC. `null` on `Autocompound Update`.
        </ResponseField>

        <ResponseField name="token_operation" type="string | null">
          Set on `Autocompound Update` records (e.g. `"3"`); otherwise `null`
        </ResponseField>

        <ResponseField name="claimed_amount_wei" type="string | null">
          Amount claimed, in the smallest unit of `token`. Set on `Claim` records.
        </ResponseField>

        <ResponseField name="drg_amount_wei" type="string | null">
          DRG amount compounded. Set on `Compound` and `Autocompound` records.
        </ResponseField>

        <ResponseField name="usdc_amount_wei" type="string | null">
          USDC amount compounded. Set on `Compound` and `Autocompound` records.
        </ResponseField>

        <ResponseField name="tx_hash" type="string">
          Transaction hash
        </ResponseField>

        <ResponseField name="timestamp" type="string">
          Unix timestamp in seconds, as a string
        </ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="staking_activity" type="object[]">
      Deposits and withdrawals

      <Expandable title="Staking activity properties">
        <ResponseField name="type" type="string">
          `"Deposit"`, `"Withdraw Start"`, or `"Instant Withdraw"`
        </ResponseField>

        <ResponseField name="user" type="string">
          User wallet address
        </ResponseField>

        <ResponseField name="amount_wei" type="string">
          DRG amount
        </ResponseField>

        <ResponseField name="fees" type="string | null">
          Fee value in DRG's smallest unit. Set on `Instant Withdraw` records; otherwise `null`.
        </ResponseField>

        <ResponseField name="tx_hash" type="string">
          Transaction hash
        </ResponseField>

        <ResponseField name="timestamp" type="string">
          Unix timestamp in seconds, as a string
        </ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="marketplace_activity" type="object[]">
      Staking-position marketplace actions

      <Expandable title="Marketplace activity properties">
        <ResponseField name="type" type="string">
          `"Listed"`, `"Price updated"`, `"Removed"`, `"Bought"`, or `"Sold"`
        </ResponseField>

        <ResponseField name="user" type="string">
          User wallet address
        </ResponseField>

        <ResponseField name="amount_wei" type="string | null">
          DRG amount of the position. `null` on `Removed`.
        </ResponseField>

        <ResponseField name="price_in_USDC_wei" type="string | null">
          Price in USDC's smallest unit (6 decimals). `null` on `Removed`.
        </ResponseField>

        <ResponseField name="bought_from" type="string | null">
          Seller address. Set on `Bought` records.
        </ResponseField>

        <ResponseField name="sold_to" type="string | null">
          Buyer address. Set on `Sold` records.
        </ResponseField>

        <ResponseField name="tx_hash" type="string">
          Transaction hash
        </ResponseField>

        <ResponseField name="timestamp" type="string">
          Unix timestamp in seconds, as a string
        </ResponseField>
      </Expandable>
    </ResponseField>

    <ResponseField name="staking_bonus" type="object[]">
      Staking bonus payouts

      <Expandable title="Staking bonus properties">
        <ResponseField name="type" type="string">
          `"Bonus"`
        </ResponseField>

        <ResponseField name="user" type="string">
          User wallet address
        </ResponseField>

        <ResponseField name="token" type="string">
          Token address of the bonus (currently the staking contract address)
        </ResponseField>

        <ResponseField name="amount" type="string">
          Human-readable bonus amount, as a decimal string
        </ResponseField>

        <ResponseField name="amount_wei" type="string">
          Bonus amount in the token's smallest unit (18 decimals)
        </ResponseField>

        <ResponseField name="amount_usd_value" type="string">
          USD value of the bonus, as a decimal string
        </ResponseField>

        <ResponseField name="tx_hash" type="string">
          Transaction hash
        </ResponseField>

        <ResponseField name="timestamp" type="string">
          Unix timestamp in seconds, as a string
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>
