---
title: "Staking Rewards"
description: "Protocol rewards, withdrawal fees, and marketplace fees"
api: "GET https://sei-api.dragonswap.app/api/v1/staking/{address}/rewards"
---

<Note>
  The `{address}` path segment is currently not used to filter results; any value returns the same data.
</Note>

All arrays are sorted newest first and are not paginated.

## Path Parameters

<ParamField path="address" type="string" required default="0xb71886c52d754cf2b1d3d866c7cfe9eec01418a7">
  Staking contract address
</ParamField>

## Response Fields

<ResponseField name="status" type="string">
  `"ok"` on success
</ResponseField>

<ResponseField name="fees" type="object">
  Aggregated totals

  <Expandable title="Fees object properties">
    <ResponseField name="total_withdrawal_fees_drg" type="number">
      Aggregate withdrawal fees. Despite the `_drg` suffix, this does not equal the sum of `drg_fee` in `withdrawl_fees`; it approximately matches their USD value.
    </ResponseField>

    <ResponseField name="total_marketplace_fees_drg" type="number">
      Aggregate marketplace fees. Despite the `_drg` suffix, this does not equal the sum of `drg_fee` in `marketplace_fees`; it approximately matches their USD value.
    </ResponseField>

    <ResponseField name="total_protocol_rewards_usd" type="number">
      Total USD value of `protocol_rewards` (sum of `amount` × `token_usd_price`)
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="withdrawl_fees" type="object[]">
  Withdrawal fee records. The key is spelled `withdrawl_fees` in the API.

  <Expandable title="Withdrawal fee object properties">
    <ResponseField name="tx_hash" type="string">
      Transaction hash
    </ResponseField>

    <ResponseField name="created_at" type="string">
      Unix timestamp in seconds, as a string
    </ResponseField>

    <ResponseField name="drg_fee" type="number">
      Fee amount in DRG (human-readable)
    </ResponseField>

    <ResponseField name="reward_per_usd_staked" type="string">
      Reward per USD staked, as a decimal string (may use exponent notation, e.g. `"8.1500662470371e-05"`)
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="marketplace_fees" type="object[]">
  Marketplace fee records

  <Expandable title="Marketplace fee object properties">
    <ResponseField name="tx_hash" type="string">
      Transaction hash
    </ResponseField>

    <ResponseField name="created_at" type="string">
      Unix timestamp in seconds, as a string
    </ResponseField>

    <ResponseField name="drg_fee" type="number">
      Fee amount in DRG (human-readable)
    </ResponseField>

    <ResponseField name="reward_per_usd_staked" type="string">
      Reward per USD staked, as a decimal string (may use exponent notation)
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="protocol_rewards" type="object[]">
  Protocol reward emissions. Same records as `deposits` in [Staking Emissions](/contracts/dex-api/staking/emissions).

  <Expandable title="Protocol reward object properties">
    <ResponseField name="tx_hash" type="string">
      Transaction hash
    </ResponseField>

    <ResponseField name="token_address" type="string">
      Reward token address: DRG (`0x0a526e425809aea71eb279d24ae22dee6c92a4fe`) or USDC (`0xe15fc38f6d8c56af07bbcbe3baf5708a2bf42392`)
    </ResponseField>

    <ResponseField name="amount" type="number">
      Human-readable reward amount
    </ResponseField>

    <ResponseField name="amount_wei" type="string">
      Reward amount in the token's smallest unit (18 decimals for DRG, 6 for USDC)
    </ResponseField>

    <ResponseField name="created_at" type="string">
      Unix timestamp in seconds, as a string
    </ResponseField>

    <ResponseField name="apr" type="number">
      Staking APR at the time of the reward, in percent
    </ResponseField>

    <ResponseField name="reward_per_usd_staked" type="number">
      Reward per USD staked
    </ResponseField>

    <ResponseField name="token_usd_price" type="number">
      USD price of the reward token at the time of the reward
    </ResponseField>

    <ResponseField name="buy_tx_hash" type="string | null">
      Transaction hash of the token buy, if applicable
    </ResponseField>
  </Expandable>
</ResponseField>
