---
title: "Pool Activity"
description: "Paginated pool activity using block number cursors"
api: "GET https://sei-api.dragonswap.app/api/v1/pools/{address}/activity"
---

## Path Parameters

<ParamField path="address" type="string" required default="0x4cea29353508b81b6efbfb8d089efe808e8fc991">
  Pool contract address (case-insensitive).
</ParamField>

## Query Parameters

<ParamField query="swaps_from_block" type="number" default="0">
  Block number cursor for swap events. Returns swaps with `block_number` greater than or equal to this value.
</ParamField>

<ParamField query="mints_from_block" type="number" default="0">
  Block number cursor for mint (add liquidity) events. Returns mints with `block_number` greater than or equal to this value.
</ParamField>

<ParamField query="burns_from_block" type="number" default="0">
  Block number cursor for burn (remove liquidity) events. Returns burns with `block_number` greater than or equal to this value.
</ParamField>

<Info>
  Each array holds at most 500 events, sorted oldest first (`block_number` ascending). To fetch the next page, pass the `block_number` of the last event as the cursor. The cursor is inclusive, so events from that block are returned again on the next page; skip entries you already have (for example by `tx_hash`).
</Info>

## Response Fields

<ResponseField name="status" type="string">
  Request status indicator: `ok` on success, `error` when the pool does not exist (HTTP 404).
</ResponseField>

<ResponseField name="swaps" type="array">
  Array of swap activity objects, oldest first.

  <Expandable title="Swap object">
    <ResponseField name="pool_address" type="string">
      Address of the pool where the swap occurred.
    </ResponseField>
    <ResponseField name="first_token_address" type="string">
      Address of the first token in the swap. This is not always the pool's `token0`; it changes from swap to swap together with `action`.
    </ResponseField>
    <ResponseField name="second_token_address" type="string">
      Address of the second token in the swap.
    </ResponseField>
    <ResponseField name="first_amount_token" type="number">
      Amount of the first token swapped, in whole token units (decimal-adjusted, not wei).
    </ResponseField>
    <ResponseField name="second_amount_token" type="number">
      Amount of the second token swapped, in whole token units (decimal-adjusted, not wei).
    </ResponseField>
    <ResponseField name="block_number" type="number">
      Block number in which the transaction was included.
    </ResponseField>
    <ResponseField name="tx_hash" type="string">
      Transaction hash.
    </ResponseField>
    <ResponseField name="from" type="string">
      Address that initiated the swap.
    </ResponseField>
    <ResponseField name="total_value" type="number">
      Total value of the swap in USD.
    </ResponseField>
    <ResponseField name="type" type="string">
      Transaction type: `Swap`.
    </ResponseField>
    <ResponseField name="action" type="string">
      Swap direction: `Buy` or `Sell`.
    </ResponseField>
    <ResponseField name="created_at" type="number">
      Time of the transaction, as a Unix timestamp (seconds).
    </ResponseField>
    <ResponseField name="pool_type" type="string">
      Pool version: `v3` or `v2`. `v2` = DEX V1 (constant-product) pool; `v3` = DEX V2 (concentrated-liquidity) pool.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="mints" type="array">
  Array of mint (add liquidity) events selected by `mints_from_block`. Currently always returned empty.
</ResponseField>

<ResponseField name="burns" type="array">
  Array of burn (remove liquidity) events selected by `burns_from_block`. Currently always returned empty.
</ResponseField>
