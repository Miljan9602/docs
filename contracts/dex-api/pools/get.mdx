---
title: "Get Pool"
description: "Returns details for a single pool"
api: "GET https://sei-api.dragonswap.app/api/v1/pools/{address}"
---

## Path Parameters

<ParamField path="address" type="string" required default="0x4cea29353508b81b6efbfb8d089efe808e8fc991">
  Pool contract address (case-insensitive).
</ParamField>

## Response Fields

<ResponseField name="status" type="string">
  Request status indicator: `ok` on success, `error` when the pool does not exist (HTTP 404).
</ResponseField>

<ResponseField name="pools" type="object">
  Pool details object. The key is plural (`pools`) even though it holds a single object.

  <Expandable title="Pool object">
    <ResponseField name="pool_address" type="string">
      Contract address of the pool.
    </ResponseField>
    <ResponseField name="token0_address" type="string">
      Address of the first token in the pair.
    </ResponseField>
    <ResponseField name="token1_address" type="string">
      Address of the second token in the pair.
    </ResponseField>
    <ResponseField name="first_token_reserve" type="number">
      Reserve of `token0`, in whole token units (decimal-adjusted, not wei).
    </ResponseField>
    <ResponseField name="second_token_reserve" type="number">
      Reserve of `token1`, in whole token units (decimal-adjusted, not wei).
    </ResponseField>
    <ResponseField name="liquidity" type="number">
      Total pool liquidity in USD.
    </ResponseField>
    <ResponseField name="daily_volume" type="number">
      Trading volume over the last 24 hours, in USD.
    </ResponseField>
    <ResponseField name="daily_fees" type="number">
      Fees collected over the last 24 hours, in USD.
    </ResponseField>
    <ResponseField name="hourly_volume" type="number | null">
      Trading volume over the last 60 minutes, in USD.
    </ResponseField>
    <ResponseField name="hourly_fees" type="number | null">
      Fees collected over the last 60 minutes, in USD.
    </ResponseField>
    <ResponseField name="hourly_swaps_count" type="number | null">
      Number of swaps over the last 60 minutes.
    </ResponseField>
    <ResponseField name="type" type="string">
      Pool type: `V3_POOL` or `V2_POOL`. `V2_POOL` = DEX V1 (constant-product) pool; `V3_POOL` = DEX V2 (concentrated-liquidity) pool.
    </ResponseField>
    <ResponseField name="fee_tier" type="number">
      Swap fee as a percentage, e.g. `0.3` = 0.3%.
    </ResponseField>
    <ResponseField name="booster_farm_address" type="string | null">
      Address of the associated booster farm, if any.
    </ResponseField>
    <ResponseField name="incentive_id" type="string | null">
      Associated incentive identifier, if any.
    </ResponseField>
    <ResponseField name="apy_booster" type="number">
      APY boost from the booster farm.
    </ResponseField>
    <ResponseField name="apr" type="number">
      Annual percentage rate for the pool, in percent, based on the last 24 hours of fees (`daily_fees × 365 / liquidity × 100`).
    </ResponseField>
    <ResponseField name="pool_created_at" type="number">
      Pool creation time, as a Unix timestamp (seconds).
    </ResponseField>
    <ResponseField name="base_apr" type="number">
      Base APR, in percent.
    </ResponseField>
    <ResponseField name="boosted_apr" type="number">
      Boosted APR, in percent.
    </ResponseField>
    <ResponseField name="liquidity_change_24h" type="number | null">
      Percentage change in `liquidity` over the last 24 hours.
    </ResponseField>
    <ResponseField name="volume_change_24h" type="number | null">
      Percentage change in 24-hour trading volume compared with 24 hours earlier.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="incentives" type="array">
  Array of incentive programs associated with this pool, including programs that have already ended. Compare `end_time` with the current time to find active programs.

  <Expandable title="Incentive object">
    <ResponseField name="start_time" type="number">
      Incentive program start time, as a Unix timestamp (seconds).
    </ResponseField>
    <ResponseField name="end_time" type="number">
      Incentive program end time, as a Unix timestamp (seconds).
    </ResponseField>
    <ResponseField name="reward" type="number">
      Total reward amount.
    </ResponseField>
    <ResponseField name="reward_token" type="string">
      Address of the reward token.
    </ResponseField>
    <ResponseField name="refundee" type="string">
      Address that receives unclaimed rewards.
    </ResponseField>
    <ResponseField name="pool" type="string">
      Pool address the incentive applies to.
    </ResponseField>
    <ResponseField name="incentive_id" type="string">
      Unique identifier of the incentive.
    </ResponseField>
    <ResponseField name="apr" type="string">
      Annual percentage rate of the incentive, in percent, as a numeric string.
    </ResponseField>
    <ResponseField name="title" type="string | null">
      Display title of the incentive program.
    </ResponseField>
    <ResponseField name="liquidity_usd" type="string">
      Liquidity participating in the incentive, in USD, as a numeric string.
    </ResponseField>
    <ResponseField name="owner" type="string | null">
      Address of the incentive creator.
    </ResponseField>
    <ResponseField name="campaign" type="string | null">
      Campaign identifier.
    </ResponseField>
  </Expandable>
</ResponseField>
