---
title: "Get Farm"
description: "Returns details for a single farm"
api: "GET https://sei-api.dragonswap.app/api/v1/farms/{address}"
---

## Path Parameters

<ParamField path="address" type="string" required default="0x256df69a9a1e7ccd48ffc4211fcaba39a56635a9">
  Farm contract address (case-insensitive)
</ParamField>

## Response Fields

<ResponseField name="status" type="string">
  `"ok"` on success
</ResponseField>

<ResponseField name="tokens" type="object[]">
  The farm's reward token and, for `boosted` farms, its booster token. Each has `address`, `name`, `symbol`, `usd_price` (number) and `decimals` (number).
</ResponseField>

<ResponseField name="farm" type="object">
  Farm details. Same fields as a farm object from [List Farms](/contracts/dex-api/farms/list), except `is_funded` (not returned here), plus `info`.

  <Expandable title="Farm object properties">
    <ResponseField name="farm_address" type="string">
      Farm contract address
    </ResponseField>

    <ResponseField name="pooled_token_address" type="string">
      Address of the staked token (the pool's LP token)
    </ResponseField>

    <ResponseField name="reward_token_address" type="string">
      Reward token address
    </ResponseField>

    <ResponseField name="booster_token_address" type="string | null">
      Booster token address. `null` for `basic` farms.
    </ResponseField>

    <ResponseField name="type" type="string">
      Farm type: `"basic"` or `"boosted"`
    </ResponseField>

    <ResponseField name="apy" type="number">
      Annual percentage yield, in percent
    </ResponseField>

    <ResponseField name="tvl" type="number">
      Total value locked in USD
    </ResponseField>

    <ResponseField name="unlock_rate" type="number">
      Reward tokens unlocked per second
    </ResponseField>

    <ResponseField name="booster_unlock_rate" type="number | null">
      Booster tokens unlocked per second. `null` for `basic` farms.
    </ResponseField>

    <ResponseField name="is_lp_pooled_token" type="boolean">
      Whether the staked token is an LP token
    </ResponseField>

    <ResponseField name="end_timestamp" type="number">
      Farm end time, Unix timestamp (seconds)
    </ResponseField>

    <ResponseField name="start_timestamp" type="number">
      Farm start time, Unix timestamp (seconds)
    </ResponseField>

    <ResponseField name="total_rewards_usd" type="number">
      Total rewards value in USD
    </ResponseField>

    <ResponseField name="campaign" type="string | null">
      Campaign name, e.g. `"DS Bonus Epoch"`
    </ResponseField>

    <ResponseField name="info" type="string | null">
      Additional farm description as an HTML string (e.g. a `<ul>` list), or `null`
    </ResponseField>
  </Expandable>
</ResponseField>
