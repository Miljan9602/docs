---
title: "User Farms"
description: "All farms where the user has a non-zero balance"
api: "GET https://sei-api.dragonswap.app/api/v1/user/{address}/farms"
---

## Path Parameters

<ParamField path="address" type="string" required>
  User wallet address (case-insensitive). A value that is not a valid address returns `200` with empty arrays.
</ParamField>

## Response Fields

<ResponseField name="status" type="string">
  `"ok"` on success
</ResponseField>

<ResponseField name="tokens" type="array">
  The reward and booster tokens of the farms in `farms`. Each entry includes `address`, `name`, `symbol`, `usd_price`, and `decimals`.
</ResponseField>

<ResponseField name="pools" type="array">
  The pools whose LP tokens are staked in the farms (one per `pooled_token_address`), with the same market fields as [List Pools](/contracts/dex-api/pools/list). `type` is `"V2_POOL"` or `"V3_POOL"`: `V2_POOL` = DEX V1 (constant-product) pool; `V3_POOL` = DEX V2 (concentrated-liquidity) pool.
</ResponseField>

<ResponseField name="farms" type="array">
  Farms where the user has a non-zero balance. The response does not include the size of the user's balance; use [User Single Farm](/contracts/dex-api/user/farm) history or `farm_balance_usd` from [User Portfolio](/contracts/dex-api/user/portfolio).

  <Expandable title="Farm object">
    <ResponseField name="farm_address" type="string">
      Farm contract address.
    </ResponseField>
    <ResponseField name="pooled_token_address" type="string">
      Address of the token staked in the farm (an LP token when `is_lp_pooled_token` is `true`).
    </ResponseField>
    <ResponseField name="reward_token_address" type="string">
      Reward token address.
    </ResponseField>
    <ResponseField name="booster_token_address" type="string | null">
      Booster reward token address, if any.
    </ResponseField>
    <ResponseField name="apy" type="number">
      Annual percentage yield.
    </ResponseField>
    <ResponseField name="tvl" type="number">
      Total value locked in USD.
    </ResponseField>
    <ResponseField name="type" type="string">
      Farm type: `"basic"` or `"boosted"`.
    </ResponseField>
    <ResponseField name="unlock_rate" type="number">
      Reward tokens emitted per second.
    </ResponseField>
    <ResponseField name="booster_unlock_rate" type="number | null">
      Booster tokens emitted per second.
    </ResponseField>
    <ResponseField name="is_lp_pooled_token" type="boolean">
      Whether the staked token is an LP token.
    </ResponseField>
    <ResponseField name="start_timestamp" type="number">
      Farm start time, as a Unix timestamp (seconds).
    </ResponseField>
    <ResponseField name="end_timestamp" type="number">
      Farm end time, as a Unix timestamp (seconds).
    </ResponseField>
    <ResponseField name="total_rewards_usd" type="number">
      Total rewards value in USD.
    </ResponseField>
    <ResponseField name="campaign" type="string | null">
      Campaign identifier, if any.
    </ResponseField>
    <ResponseField name="is_funded" type="boolean | null">
      Whether the farm is funded.
    </ResponseField>
  </Expandable>
</ResponseField>
