---
title: "User Pools"
description: "A user's liquidity positions across V2 and V3 pools"
api: "GET https://sei-api.dragonswap.app/api/v1/user/{address}/pools"
---

## Path Parameters

<ParamField path="address" type="string" required>
  User wallet address (case-insensitive). A value that is not a valid address returns `200` with empty arrays.
</ParamField>

## Response Fields

<ResponseField name="status" type="string">
  `"ok"` on success
</ResponseField>

<ResponseField name="tokens" type="array">
  Tokens on both sides of every pool in `pools`.

  <Expandable title="Token object">
    <ResponseField name="address" type="string">
      Token contract address.
    </ResponseField>
    <ResponseField name="name" type="string">
      Full token name.
    </ResponseField>
    <ResponseField name="symbol" type="string">
      Token ticker symbol.
    </ResponseField>
    <ResponseField name="usd_price" type="number">
      Current USD price of the token.
    </ResponseField>
    <ResponseField name="decimals" type="number">
      Number of decimals the token uses.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="pools" type="array">
  The user's liquidity positions. `V2_POOL` = DEX V1 (constant-product) pool; `V3_POOL` = DEX V2 (concentrated-liquidity) pool. A `V2_POOL` entry is the user's LP-token position in that pool. Each `V3_POOL` position is returned as its own entry, so the same `pool_address` can appear more than once; entries do not include a position token ID or tick range.

  <Expandable title="Pool position object">
    <ResponseField name="pool_address" type="string">
      Contract address of the pool.
    </ResponseField>
    <ResponseField name="token0_address" type="string">
      Address of the first token in the pair.
    </ResponseField>
    <ResponseField name="token1_address" type="string">
      Address of the second token in the pair.
    </ResponseField>
    <ResponseField name="type" type="string">
      Pool type: `"V2_POOL"` or `"V3_POOL"`.
    </ResponseField>
    <ResponseField name="fee_tier" type="number">
      Pool fee in percent (e.g. `0.3` = 0.3%).
    </ResponseField>
    <ResponseField name="lp_balance" type="number">
      The user's LP-token balance. Present on `V2_POOL` entries only; the key is absent on `V3_POOL` entries.
    </ResponseField>
    <ResponseField name="first_token_amount" type="number">
      Amount of `token0` in the user's position, in token units (not wei).
    </ResponseField>
    <ResponseField name="second_token_amount" type="number">
      Amount of `token1` in the user's position, in token units (not wei).
    </ResponseField>
    <ResponseField name="total_liquidity_usd" type="number">
      USD value of the user's position.
    </ResponseField>
    <ResponseField name="first_token_reserve" type="number">
      Pool reserve of `token0`, in token units.
    </ResponseField>
    <ResponseField name="second_token_reserve" type="number">
      Pool reserve of `token1`, in token units.
    </ResponseField>
    <ResponseField name="liquidity" type="number">
      Total pool liquidity in USD.
    </ResponseField>
    <ResponseField name="daily_volume" type="number">
      24-hour pool trading volume in USD.
    </ResponseField>
    <ResponseField name="daily_fees" type="number">
      24-hour pool fees in USD.
    </ResponseField>
    <ResponseField name="hourly_volume" type="number | null">
      Last-hour pool trading volume in USD, or `null` when no hourly data is available.
    </ResponseField>
    <ResponseField name="hourly_fees" type="number | null">
      Last-hour pool fees in USD, or `null` when no hourly data is available.
    </ResponseField>
    <ResponseField name="hourly_swaps_count" type="number | null">
      Last-hour swap count, or `null` when no hourly data is available.
    </ResponseField>
    <ResponseField name="apr" type="number">
      Pool APR in percent.
    </ResponseField>
    <ResponseField name="base_apr" type="number">
      Base pool APR in percent.
    </ResponseField>
    <ResponseField name="boosted_apr" type="number">
      Boosted pool APR in percent.
    </ResponseField>
    <ResponseField name="apy_booster" type="number">
      APY boost from the booster farm.
    </ResponseField>
    <ResponseField name="booster_farm_address" type="string | null">
      Address of the associated booster farm, if any.
    </ResponseField>
    <ResponseField name="incentive_id" type="string | null">
      Associated incentive identifier, if any.
    </ResponseField>
    <ResponseField name="pool_created_at" type="number | null">
      Pool creation time, as a Unix timestamp (seconds).
    </ResponseField>
    <ResponseField name="liquidity_change_24h" type="number | null">
      24-hour change in pool liquidity, or `null`.
    </ResponseField>
    <ResponseField name="volume_change_24h" type="number | null">
      24-hour change in pool volume, or `null`.
    </ResponseField>
  </Expandable>
</ResponseField>

<ResponseField name="incentives" type="array">
  Incentive programs linked to the user's pools. Empty array when there are none.
</ResponseField>
