---
title: "Get Token"
description: "Returns details for a single token"
api: "GET https://sei-api.dragonswap.app/api/v1/tokens/{address}"
---

## Path Parameters

<ParamField path="address" type="string" required default="0xe30fedd158a2e3b13e9badaeabafc5516e95e8c7">
  Token contract address (case-insensitive).
</ParamField>

## Response

<ResponseField name="status" type="string">
  Request status indicator: `ok` on success, `fail` when the token is not found (HTTP 404).
</ResponseField>

<ResponseField name="token" type="object">
  Token details object.

  <Expandable title="Token object">
    <ResponseField name="address" type="string">
      Token contract address.
    </ResponseField>
    <ResponseField name="name" type="string">
      Token name.
    </ResponseField>
    <ResponseField name="symbol" type="string">
      Token ticker symbol.
    </ResponseField>
    <ResponseField name="usd_price" type="number">
      Current USD price.
    </ResponseField>
    <ResponseField name="decimals" type="number">
      Token decimal places.
    </ResponseField>
    <ResponseField name="liquidity" type="number">
      Total liquidity in USD.
    </ResponseField>
    <ResponseField name="daily_volume" type="number">
      Trading volume over the last 24 hours in USD.
    </ResponseField>
    <ResponseField name="daily_transactions" type="number">
      Cumulative transaction count. Despite the name, it is not limited to the last 24 hours.
    </ResponseField>
    <ResponseField name="verified" type="boolean">
      `true` if the token is on the [approved token list](/contracts/dex-api/tokens/approved).
    </ResponseField>
    <ResponseField name="category" type="string | null">
      Token category. Observed values: `wrapped`, `stablecoin`, `defi`, `lst`, `meme`, `native`, `ai`.
    </ResponseField>
    <ResponseField name="description" type="string | null">
      Token description, if available.
    </ResponseField>
    <ResponseField name="website" type="string | null">
      Project website URL, if available.
    </ResponseField>
    <ResponseField name="socials" type="object | null">
      Social links, if available. Only the keys that have a link are present.

      <Expandable title="Socials object">
        <ResponseField name="x" type="string">
          X (Twitter) URL.
        </ResponseField>
        <ResponseField name="telegram" type="string">
          Telegram URL.
        </ResponseField>
        <ResponseField name="discord" type="string">
          Discord URL.
        </ResponseField>
        <ResponseField name="github" type="string">
          GitHub URL.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="liquidity_change_24h" type="number | null">
      Percentage change in `liquidity` over the last 24 hours.
    </ResponseField>
    <ResponseField name="volume_change_24h" type="number | null">
      Percentage change in 24-hour trading volume compared with 24 hours earlier.
    </ResponseField>
  </Expandable>
</ResponseField>
