---
title: "List Tokens"
description: "Returns all tokens with liquidity above the minimum TVL threshold"
api: "GET https://sei-api.dragonswap.app/api/v1/tokens"
---

## Response

<ResponseField name="status" type="string">
  Request status indicator.
</ResponseField>

<ResponseField name="tokens" type="array">
  Array of token objects.

  <Expandable title="Token object">
    <ResponseField name="address" type="string">
      Token contract address.
    </ResponseField>
    <ResponseField name="name" type="string">
      Token name.
    </ResponseField>
    <ResponseField name="symbol" type="string">
      Token ticker symbol.
    </ResponseField>
    <ResponseField name="usd_price" type="number">
      Current USD price.
    </ResponseField>
    <ResponseField name="decimals" type="number">
      Token decimal places.
    </ResponseField>
    <ResponseField name="liquidity" type="number">
      Total liquidity in USD.
    </ResponseField>
    <ResponseField name="description" type="string | null">
      Token description, if available.
    </ResponseField>
    <ResponseField name="verified" type="boolean">
      `true` if the token is on the [approved token list](/contracts/dex-api/tokens/approved).
    </ResponseField>
    <ResponseField name="category" type="string | null">
      Token category. Observed values: `wrapped`, `stablecoin`, `defi`, `lst`, `meme`, `native`, `ai`.
    </ResponseField>
    <ResponseField name="change" type="object">
      Price change percentages over various timeframes.

      <Expandable title="Change object">
        <ResponseField name="hourly" type="number">
          Hourly price change percentage.
        </ResponseField>
        <ResponseField name="daily" type="number">
          Daily price change percentage.
        </ResponseField>
        <ResponseField name="weekly" type="number">
          Weekly price change percentage.
        </ResponseField>
        <ResponseField name="monthly" type="number">
          Monthly price change percentage.
        </ResponseField>
      </Expandable>
    </ResponseField>
    <ResponseField name="liquidity_change_24h" type="number | null">
      Percentage change in `liquidity` over the last 24 hours.
    </ResponseField>
    <ResponseField name="volume_change_24h" type="number | null">
      Percentage change in 24-hour trading volume compared with 24 hours earlier.
    </ResponseField>
    <ResponseField name="sparkline_7d" type="array">
      Daily USD prices for the last 7 days (7 entries, oldest first; the last entry is the current day).

      <Expandable title="Sparkline entry">
        <ResponseField name="day" type="string">
          Date in `YYYY-MM-DD` format.
        </ResponseField>
        <ResponseField name="price" type="number">
          USD price for that day.
        </ResponseField>
      </Expandable>
    </ResponseField>
  </Expandable>
</ResponseField>
